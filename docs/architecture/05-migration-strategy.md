Future<bool> validateDataConsistency() async {
  final supabaseEvents = await VEventsTable().queryRows(...);
  final convertedEvents = supabaseEvents.map((row) => 
    functions.convertVEventsListToEventStruct(row)
  ).toList();
  
  // 驗證轉換後資料完整性
